    // ============ 桌面 panel: This PC + Recycle Bin + archived desktop + live desktop ============
    class DesktopForm : Form
    {
        TextBox box;
        ScrollPanel2 content;
        Label status;
        List<DTile> tiles = new List<DTile>();
        List<DSection> sections = new List<DSection>();
        class DTile { public string Name; public string Path; public bool IsDir; public string OpenArg; public Bitmap Icon; public Control Ctl; }
        class DSection { public string Name; public Label Head; public FlowPanel2 Host; public List<DTile> Items = new List<DTile>(); }
        class ScrollPanel2 : Panel { public ScrollPanel2() { DoubleBuffered = true; BackColor = Color.Transparent; AutoScroll = true; } }
        class FlowPanel2 : FlowLayoutPanel { public FlowPanel2() { DoubleBuffered = true; BackColor = Color.Transparent; AutoSize = true; AutoSizeMode = AutoSizeMode.GrowAndShrink; WrapContents = true; } }

        public static void ShowDesk()
        {
            try
            {
                DesktopForm f = new DesktopForm();
                f.Show();
                f.Activate();
            }
            catch (Exception ex) { Program.Log("[DESK] open fail " + ex.Message); }
        }

        public DesktopForm()
        {
            FormBorderStyle = FormBorderStyle.None;
            ShowInTaskbar = false;
            StartPosition = FormStartPosition.Manual;
            TopMost = true;
            KeyPreview = true;
            Rectangle sc = Screen.PrimaryScreen.Bounds;
            int m = Math.Max(8, sc.Height / 40);
            Bounds = new Rectangle(m, m, sc.Width - 2 * m, sc.Height - 2 * m);
            BackColor = Color.FromArgb(16, 21, 36);
            DoubleBuffered = true;

            Label title = new Label();
            title.Text = "桌面";
            title.ForeColor = Color.FromArgb(165, 150, 180, 215);
            title.Font = new Font("Microsoft YaHei UI", 11f);
            title.AutoSize = true;
            title.Location = new Point(34, 26);
            Controls.Add(title);

            Label close = new Label();
            close.Text = "✕  关闭";
            close.ForeColor = Color.FromArgb(175, 155, 170, 195);
            close.Font = new Font("Microsoft YaHei UI", 10f);
            close.AutoSize = true;
            close.Location = new Point(Width - 116, 26);
            close.Cursor = Cursors.Hand;
            close.Click += delegate { Close(); };
            Controls.Add(close);

            box = new TextBox();
            box.BorderStyle = BorderStyle.None;
            box.BackColor = Color.FromArgb(255, 28, 35, 56);
            box.ForeColor = Color.White;
            box.Font = new Font("Microsoft YaHei UI", 15f);
            box.Bounds = new Rectangle(Width / 2 - 340, 58, 680, 46);
            box.TextChanged += delegate { ApplyFilter(); };
            Controls.Add(box);

            status = new Label();
            status.Text = " 正在读取桌面内容…";
            status.ForeColor = Color.FromArgb(140, 135, 155, 185);
            status.Font = new Font("Microsoft YaHei UI", 9f);
            status.AutoSize = true;
            status.Location = new Point(Width / 2 - 340, 112);
            Controls.Add(status);

            content = new ScrollPanel2();
            content.Bounds = new Rectangle(28, 138, Width - 56, Height - 164);
            Controls.Add(content);

            Paint += delegate(object s, PaintEventArgs e)
            {
                using (Pen p = new Pen(Color.FromArgb(80, 96, 210, 255), 1.3f))
                    e.Graphics.DrawRectangle(p, 0, 0, Width - 1, Height - 1);
                using (Pen p2 = new Pen(Color.FromArgb(60, 96, 150, 210), 1f))
                    e.Graphics.DrawRectangle(p2, box.Left - 12, box.Top - 9, box.Width + 24, box.Height + 16);
            };
            Deactivated += delegate { Close(); };
            KeyDown += delegate(object s, KeyEventArgs e)
            {
                if (e.KeyCode == Keys.Escape) Close();
            };

            Shown += delegate
            {
                box.Focus();
                ThreadPool.QueueUserWorkItem(delegate { Populate(); });
            };
        }

        void BuildAcrylicBackground()
        {
            try
            {
                Image result = null;
                using (Bitmap raw = new Bitmap(Width, Height))
                {
                    using (Graphics g = Graphics.FromImage(raw))
                        g.CopyFromScreen(Left, Top, 0, 0, raw.Size);
                    int sw = Math.Max(1, Width / 10), sh = Math.Max(1, Height / 10);
                    using (Bitmap small = new Bitmap(raw, sw, sh))
                    {
                        for (int i = 0; i < 3; i++) LauncherForm.BoxBlur(small);
                        using (Bitmap bg = new Bitmap(small, Width, Height))
                        using (Graphics g = Graphics.FromImage(bg))
                        {
                            using (SolidBrush tint = new SolidBrush(Color.FromArgb(178, 14, 18, 32)))
                                g.FillRectangle(tint, 0, 0, Width, Height);
                            result = (Image)bg.Clone();
                        }
                    }
                }
                if (result != null)
                    SafeInvoke(delegate { if (!IsDisposed) { Image old = BackgroundImage; BackgroundImage = result; if (old != null) old.Dispose(); } Invalidate(); });
            }
            catch (Exception ex) { Program.Log("[DESK] acrylic fail " + ex.Message); }
        }

        void SafeInvoke(MethodInvoker d)
        {
            try { if (!IsDisposed && !Disposing) BeginInvoke(d); }
            catch { }
        }

        void ApplyFilter()
        {
            string q = (box.Text ?? "").Trim().ToLower();
            foreach (DSection s in sections)
            {
                int vis = 0;
                foreach (DTile t in s.Items)
                {
                    bool show = q.Length == 0 || t.Name.ToLower().Contains(q);
                    if (t.Ctl != null) t.Ctl.Visible = show;
                    if (show) vis++;
                }
                if (s.Head != null) s.Head.Visible = vis > 0;
                if (s.Host != null) s.Host.Visible = vis > 0;
            }
        }

        void Launch(DTile t)
        {
            try
            {
                if (t.OpenArg != null)
                    Process.Start("explorer.exe", t.OpenArg);          // shell namespace (此电脑/回收站)
                else
                    Process.Start(new ProcessStartInfo(t.Path) { UseShellExecute = true });
            }
            catch (Exception ex) { Program.Log("[DESK] open item fail " + ex.Message); }
            Close();
        }

        void Populate()
        {
            try
            {
                BuildAcrylicBackground();
                List<DTile> sys = new List<DTile>();
                DTile pc = new DTile();
                pc.Name = "此电脑"; pc.Path = "explorer.exe"; pc.IsDir = true; pc.Icon = DrawGlyph(0);
                pc.OpenArg = "shell:::{20D04FE0-3AEA-1069-A2D8-08002B30309D}";
                DTile rb = new DTile();
                rb.Name = "回收站"; rb.Path = "explorer.exe"; rb.IsDir = true; rb.Icon = DrawGlyph(1);
                rb.OpenArg = "shell:::{645FF040-5081-101B-9F08-00AA002F954E}";
                sys.Add(pc); sys.Add(rb);

                // archived desktop: newest D:\桌面归档_* folder
                string archive = null;
                try
                {
                    DateTime best = DateTime.MinValue;
                    foreach (string d in Directory.GetDirectories(@"D:\", "桌面归档_*"))
                    {
                        try
                        {
                            DateTime wt = Directory.GetLastWriteTime(d);
                            if (wt > best) { best = wt; archive = d; }
                        }
                        catch { }
                    }
                }
                catch { }
                HashSet<string> names = new HashSet<string>();
                names.Add("此电脑"); names.Add("回收站");
                List<DTile> archived = new List<DTile>();
                if (archive != null) CollectDir(archive, archived, names);

                // live desktop folders (user + common)
                List<DTile> live = new List<DTile>();
                try { CollectDir(Environment.GetFolderPath(Environment.SpecialFolder.DesktopDirectory), live, names); } catch { }
                try { CollectDir(Environment.GetFolderPath(Environment.SpecialFolder.CommonDesktopDirectory), live, names); } catch { }

                SafeInvoke(delegate
                {
                    BuildUi(sys, archive == null ? "桌面归档（未找到）" : Path.GetFileName(archive), archived, live);
                });
            }
            catch (Exception ex) { Program.Log("[DESK] populate fail " + ex.Message); }
        }

        void CollectDir(string dir, List<DTile> into, HashSet<string> names)
        {
            foreach (string p in Directory.GetFileSystemEntries(dir))
            {
                string nm = Path.GetFileName(p);
                if (nm.ToLower().EndsWith(".ini") || nm.StartsWith(".")) continue;
                if (!names.Add(nm.ToLower())) continue;
                DTile t = new DTile();
                t.Name = nm;
                t.Path = p;
                try { t.IsDir = Directory.Exists(p); } catch { }
                t.Icon = AppCache.IconFor(p);
                if (t.Icon == null && t.IsDir) t.Icon = DrawGlyph(2);
                into.Add(t);
            }
            into.Sort(delegate(DTile a, DTile b) { return string.Compare(a.Name, b.Name, StringComparison.CurrentCultureIgnoreCase); });
        }

        void BuildUi(List<DTile> sys, string archName, List<DTile> archived, List<DTile> live)
        {
            if (IsDisposed) return;
            try
            {
                AddSection("系统", sys, Color.FromArgb(255, 110, 205, 255));
                if (archived.Count > 0) AddSection(archName, archived, Color.FromArgb(200, 150, 175, 215));
                if (live.Count > 0) AddSection("当前桌面", live, Color.FromArgb(200, 150, 175, 215));
                LayoutSections();
                status.Text = "";
                ApplyFilter();
            }
            catch (Exception ex) { Program.Log("[DESK] buildui fail " + ex.Message); }
        }

        void LayoutSections()
        {
            int y = 4;
            int innerW = content.ClientSize.Width - 26;
            foreach (DSection s in sections)
            {
                s.Head.Location = new Point(8, y);
                y += 30;
                s.Host.Location = new Point(6, y);
                s.Host.MaximumSize = new Size(innerW, 0);
                s.Host.MinimumSize = new Size(innerW, 0);
                y += s.Host.PreferredSize.Height + 16;
            }
        }

        void AddSection(string name, List<DTile> items, Color headColor)
        {
            if (IsDisposed) return;
            content.SuspendLayout();
            try
            {
                DSection s = new DSection();
                s.Name = name;
                s.Items = items;
                s.Head = new Label();
                s.Head.Text = name;
                s.Head.ForeColor = headColor;
                s.Head.Font = new Font("Microsoft YaHei UI", 12.5f, FontStyle.Bold);
                s.Head.AutoSize = true;
                s.Head.BackColor = Color.Transparent;
                content.Controls.Add(s.Head);
                s.Host = new FlowPanel2();
                content.Controls.Add(s.Host);
                foreach (DTile t in items)
                {
                    if (IsDisposed) return;
                    Panel p = new Panel();
                    p.Size = new Size(102, 112);
                    p.Margin = new Padding(8, 6, 8, 6);
                    p.Cursor = Cursors.Hand;
                    p.BackColor = Color.Transparent;
                    PictureBox ic = new PictureBox();
                    ic.Size = new Size(48, 48);
                    ic.Location = new Point(27, 4);
                    ic.SizeMode = PictureBoxSizeMode.Zoom;
                    ic.BackColor = Color.Transparent;
                    if (t.Icon != null) ic.Image = t.Icon;
                    Label lb = new Label();
                    lb.Text = t.Name;
                    lb.ForeColor = Color.FromArgb(230, 232, 240);
                    lb.BackColor = Color.Transparent;
                    lb.Font = new Font("Microsoft YaHei UI", 8.5f);
                    lb.Size = new Size(100, 34);
                    lb.Location = new Point(1, 60);
                    lb.TextAlign = ContentAlignment.TopCenter;
                    p.Controls.Add(ic);
                    p.Controls.Add(lb);
                    DTile cur = t;
                    p.Click += delegate { Launch(cur); };
                    ic.Click += delegate { Launch(cur); };
                    lb.Click += delegate { Launch(cur); };
                    t.Ctl = p;
                    s.Host.Controls.Add(p);
                }
                sections.Add(s);
            }
            finally { content.ResumeLayout(false); }
        }

        // 0=monitor(此电脑) 1=bin(回收站) 2=folder
        static Bitmap DrawGlyph(int kind)
        {
            Bitmap b = new Bitmap(48, 48);
            using (Graphics g = Graphics.FromImage(b))
            {
                g.SmoothingMode = SmoothingMode.AntiAlias;
                if (kind == 0)
                {
                    using (Pen p = new Pen(Color.FromArgb(235, 225, 238, 252), 2.6f))
                    {
                        g.DrawRectangle(p, 8, 8, 32, 22);
                        g.DrawLine(p, 24, 30, 24, 36);
                        g.DrawLine(p, 15, 38, 33, 38);
                    }
                }
                else if (kind == 1)
                {
                    using (Pen p = new Pen(Color.FromArgb(160, 215, 235, 252), 2.6f))
                    {
                        g.DrawRectangle(p, 13, 12, 22, 5);            // lid
                        g.DrawLine(p, 20, 9, 28, 9);                  // handle
                        g.DrawRectangle(p, 16, 17, 16, 22);           // body
                        g.DrawLine(p, 22, 21, 22, 34);
                        g.DrawLine(p, 26, 21, 26, 34);
                    }
                }
                else
                {
                    using (SolidBrush fb = new SolidBrush(Color.FromArgb(220, 90, 160, 235)))
                        g.FillRectangle(fb, 7, 12, 34, 27);
                    using (SolidBrush fb2 = new SolidBrush(Color.FromArgb(235, 120, 185, 250)))
                        g.FillRectangle(fb2, 7, 8, 18, 8);
                }
            }
            return b;
        }
    }
